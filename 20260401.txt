WITH business_detail AS (
    SELECT
        pfc.fab_id,
        pfc.fab_name,
        a.company_code,
        a.company_name,
        a."year",
        a.product_category,
        a.inout_category AS business_category,
        'CP' AS basicType,
        ROUND(SUM(COALESCE(a.january, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS january,
        ROUND(SUM(COALESCE(a.february, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS february,
        ROUND(SUM(COALESCE(a.march, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS march,
        ROUND(SUM(COALESCE(a.april, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS april,
        ROUND(SUM(COALESCE(a.may, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS may,
        ROUND(SUM(COALESCE(a.june, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS june,
        ROUND(SUM(COALESCE(a.july, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS july,
        ROUND(SUM(COALESCE(a.august, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS august,
        ROUND(SUM(COALESCE(a.september, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS september,
        ROUND(SUM(COALESCE(a.october, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS october,
        ROUND(SUM(COALESCE(a.november, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS november,
        ROUND(SUM(COALESCE(a.december, 0) * COALESCE(b_q1.standard_cost, 0)), 8) AS december
    FROM qcmuser.qcm_business_data_total_inout a
    LEFT JOIN qcmuser.qcm_finance_data_cost_price b_q1
        ON a.year = b_q1.year AND b_q1.quarter = 'Q1' AND b_q1.archive = 0
    LEFT JOIN qcm_project_fab_config pfc 
        ON a.company_code = pfc.company_code 
        AND a.company_name = pfc.company_name 
        AND a."year" = pfc."year" 
        AND a.project = pfc.project
    WHERE
        a.year = '2025'
        AND a.archive = 0
        AND a.product_category IS NOT NULL
    GROUP BY
        a.company_code, a.company_name, a."year",
        pfc.fab_id, pfc.fab_name, a.product_category, a.inout_category
)
SELECT * FROM business_detail

UNION ALL
-- 总投入汇总（按 fab_id + fab_name 分组）
SELECT
    fab_id,
    fab_name,
    company_code,
    company_name,
    "year",
    '总投入' AS product_category,
    '投入' AS business_category,
    basicType,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN january ELSE 0 END), 8) AS january,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN february ELSE 0 END), 8) AS february,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN march ELSE 0 END), 8) AS march,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN april ELSE 0 END), 8) AS april,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN may ELSE 0 END), 8) AS may,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN june ELSE 0 END), 8) AS june,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN july ELSE 0 END), 8) AS july,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN august ELSE 0 END), 8) AS august,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN september ELSE 0 END), 8) AS september,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN october ELSE 0 END), 8) AS october,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN november ELSE 0 END), 8) AS november,
    ROUND(SUM(CASE WHEN business_category = '投入' THEN december ELSE 0 END), 8) AS december
FROM business_detail
GROUP BY 
    fab_id, 
    fab_name, 
    company_code, 
    company_name, 
    "year", 
    basicType
HAVING COUNT(1) > 0

UNION ALL
-- 总产出汇总（按 fab_id + fab_name 分组）
SELECT
    fab_id,
    fab_name,
    company_code,
    company_name,
    "year",
    '总产出' AS product_category,
    '产出' AS business_category,
    basicType,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN january ELSE 0 END), 8) AS january,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN february ELSE 0 END), 8) AS february,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN march ELSE 0 END), 8) AS march,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN april ELSE 0 END), 8) AS april,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN may ELSE 0 END), 8) AS may,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN june ELSE 0 END), 8) AS june,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN july ELSE 0 END), 8) AS july,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN august ELSE 0 END), 8) AS august,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN september ELSE 0 END), 8) AS september,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN october ELSE 0 END), 8) AS october,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN november ELSE 0 END), 8) AS november,
    ROUND(SUM(CASE WHEN business_category = '产出' THEN december ELSE 0 END), 8) AS december
FROM business_detail
GROUP BY 
    fab_id, 
    fab_name, 
    company_code, 
    company_name, 
    "year", 
    basicType
HAVING COUNT(1) > 0

ORDER BY company_code, fab_id, product_category;
